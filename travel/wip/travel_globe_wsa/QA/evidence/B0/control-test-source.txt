import test from 'node:test';
import assert from 'node:assert/strict';
import {createFlightControls} from '../travel/globe-v13/flight-controls.js';
function setup() {
 const win=new EventTarget(),doc=new EventTarget();doc.hidden=false;
 globalThis.addEventListener=win.addEventListener.bind(win);
 globalThis.removeEventListener=win.removeEventListener.bind(win);
 globalThis.document=doc;
 const controls=createFlightControls();
 const send=(type,key,repeat=false)=>{const e=new Event(type,{cancelable:true});e.key=key;e.repeat=repeat;win.dispatchEvent(e);return e;};
 return {win,doc,controls,send};
}
test('R0 movement input releases after focus loss, including missing keyup',()=>{
 const {win,controls,send}=setup();
 send('keydown','w');send('keydown','a');assert.equal(controls.getState().forward,true);assert.ok(controls.getState().turnRate>0);
 win.dispatchEvent(new Event('blur'));const s=controls.getState();assert.equal(s.forward,false);assert.equal(s.turnRate,0);
 send('keydown','s');assert.equal(controls.getState().brake,true);send('keyup','s');assert.equal(controls.getState().brake,false);
});
test('R0 hidden document releases held flight input',()=>{
 const {doc,controls,send}=setup();send('keydown','ArrowUp');assert.equal(controls.getState().elevate,true);
 doc.hidden=true;doc.dispatchEvent(new Event('visibilitychange'));assert.equal(controls.getState().elevate,false);
});
test('R0 hover and fire are consumed once; key repeat does not retrigger',()=>{
 const {controls,send}=setup();send('keydown','h');assert.equal(controls.getState().schwebeToggle,true);assert.equal(controls.getState().schwebeToggle,false);
 send('keydown','h',true);assert.equal(controls.getState().schwebeToggle,false);
 assert.equal(send('keydown',' ').defaultPrevented,true);let s=controls.getState();assert.ok(s.paintball&&s.specialAction);
 s=controls.getState();assert.equal(s.paintball,false);assert.equal(s.specialAction,false);
});
test('R0 disable clears movement and fresh input works after enabling',()=>{
 const {controls,send}=setup();send('keydown','w');controls.enabled=false;assert.equal(controls.getState().forward,false);
 send('keydown','w');controls.enabled=true;assert.equal(controls.getState().forward,false);
 send('keydown','w');assert.equal(controls.getState().forward,true);
});
